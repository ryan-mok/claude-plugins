```bash
# Get analytics directory path (always under main repo root)
# Usage: ANALYTICS_DIR=$(get_analytics_dir "$CWD")
get_analytics_dir() {
  local cwd="${1:-$CWD}"
  local root
  root=$(get_git_root "$cwd")
  echo "$root/.claude/harness/analytics"
}

# Check if running in a git worktree
# Usage: is_worktree "$CWD" && echo "yes"
is_worktree() {
  local cwd="${1:-$CWD}"
  local toplevel root
  toplevel=$(cd "$cwd" && git rev-parse --show-toplevel 2>/dev/null || echo "")
  root=$(get_git_root "$cwd")
  [[ -n "$toplevel" && "$toplevel" != "$root" ]]
}

# Emit an analytics event to events.jsonl
# Usage: emit_event "event.type" '{"extra":"fields"}' ["team"]
# Requires CWD, SESSION_PREFIX, and BRANCH to be set by caller.
# Optional 3rd arg "team" sets scope to "team" instead of "single".
emit_event() {
  local event_type="$1"
  local extra_fields="${2:-"{}"}"
  local scope="${3:-single}"
  local analytics_dir ts worktree_val repo_root merged

  analytics_dir=$(get_analytics_dir "$CWD")
  mkdir -p "$analytics_dir"

  ts=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
  repo_root=$(get_git_root "$CWD")

  if is_worktree "$CWD"; then
    worktree_val="true"
  else
    worktree_val="false"
  fi

  # Build envelope, merge with extra fields
  merged=$(jq -n -c \
    --arg ts "$ts" \
    --arg event "$event_type" \
    --arg sid "${SESSION_PREFIX:-unknown}" \
    --arg branch "${BRANCH:-unknown}" \
    --arg scope "$scope" \
    --arg repo "$repo_root" \
    --argjson wt "$worktree_val" \
    --argjson extra "$extra_fields" \
    '{ts:$ts, event:$event, session_id:$sid, branch:$branch, scope:$scope, worktree:$wt, repo_root:$repo} + $extra'
  )

  echo "$merged" >> "$analytics_dir/events.jsonl"
}
```

compute_session_end() {
  local progress_file="$1"
  local events_file="$(get_analytics_dir "$CWD")/events.jsonl"
  local loop_state="/tmp/harness-loop-state-${SESSION_PREFIX}.jsonl"

  # --- Start gh call in background ---

    timeout 8 gh pr list --head "$BRANCH" --state open --json number --jq 'length > 0' > /tmp/harness-gh-result-$$ 2>/dev/null &
    local GH_PID=$!
  fi

  # --- Read agent_outcome from progress file ---

  if [[ -f "$progress_file" ]]; then
    local after_heading
    after_heading=$(sed -n '/^## Agent Outcome/,/^##/{/^## Agent Outcome/d;/^##/d;p}' "$progress_file" | head -1 | tr -d '[:space:]')
    case "$after_heading" in
      success|partial|failed|abandoned) agent_outcome="$after_heading" ;;
    esac
  fi

  # --- Read mode from YAML frontmatter ---
  local mode="organic"
  if [[ -f "$progress_file" ]] && head -1 "$progress_file" | grep -q '^---'; then
    local fm_mode
    fm_mode=$(sed -n '/^---$/,/^---$/p' "$progress_file" | grep '^mode:' | head -1 | awk '{print $2}' | tr -d '"'"'"'')
    [[ -n "$fm_mode" ]] && mode="$fm_mode"
  fi

  # --- Read semantic_blocks ---
  local semantic_blocks=0
  if [[ -f "$progress_file" ]]; then
    semantic_blocks=$(sed -n '/^## Semantic Constraint Notes/,/^##/{/^## /d;/^- \[/p}' "$progress_file" | wc -l | tr -d ' ')
  fi

  # --- Read progress_marked_complete ---
  local progress_marked_complete="false"
  if [[ -f "$progress_file" ]]; then
    if grep -qiE '(## Status: COMPLETE|status to "Done")' "$progress_file" 2>/dev/null; then
      progress_marked_complete="true"
    elif sed -n '/^---$/,/^---$/p' "$progress_file" 2>/dev/null | grep -qiE '^status:\s*(complete|done)'; then
      progress_marked_complete="true"
    fi
  fi

  # --- Read counts from events.jsonl ---
  local loop_count=0 total_violations=0 constraint_violations_blocked=0 compaction_count=0
  if [[ -f "$events_file" ]]; then
    loop_count=$(jq -r "select(.event==\"loop.detected\" and .session_id==\"$SESSION_PREFIX\")" "$events_file" 2>/dev/null | wc -l | tr -d ' ')
    total_violations=$(jq -r "select(.event==\"constraint.violation\" and .session_id==\"$SESSION_PREFIX\")" "$events_file" 2>/dev/null | wc -l | tr -d ' ')
    constraint_violations_blocked=$(jq -r "select(.event==\"constraint.violation\" and .session_id==\"$SESSION_PREFIX\" and .decision==\"deny\")" "$events_file" 2>/dev/null | wc -l | tr -d ' ')
    compaction_count=$(jq -r "select(.event==\"session.compact\" and .session_id==\"$SESSION_PREFIX\")" "$events_file" 2>/dev/null | wc -l | tr -d ' ')
  fi

  # --- Read tests_passing from loop state ---
  local tests_passing="null"
  if [[ -f "$loop_state" ]]; then
    local last_test
    last_test=$(jq -r 'select(.tool=="Bash") | select(.file | test("\\b(test|jest|pytest|cargo test|go test|npm test|vitest|mocha|rspec)\\b"; "i"))' "$loop_state" 2>/dev/null | tail -1)
    if [[ -n "$last_test" ]]; then
      local error_field
      error_field=$(echo "$last_test" | jq -r '.error // ""')
      if [[ -n "$error_field" ]]; then
        tests_passing="false"
      else
        tests_passing="true"
      fi
    fi
  fi

  # --- Read unresolved_loops ---
  local unresolved_loops=0
  if [[ -f "$loop_state" && -f "$events_file" ]]; then
    # Get detected loop patterns from events
    local detected_patterns
    detected_patterns=$(jq -r 'select(.event=="loop.detected" and .session_id=="'"$SESSION_PREFIX"'") | "\(.tool)\t\(.file)"' "$events_file" 2>/dev/null | sort -u)
    # Check each against last 10 entries in loop state
    local recent
    recent=$(tail -10 "$loop_state")
    while IFS=$'\t' read -r dtool dfile; do
      [[ -z "$dtool" ]] && continue
      local still_present
      still_present=$(echo "$recent" | jq -r --arg t "$dtool" --arg f "$dfile" 'select(.tool==$t and .file==$f)' 2>/dev/null | wc -l | tr -d ' ')
      if [[ "$still_present" -ge 1 ]]; then
        ((unresolved_loops++))
      fi
    done <<< "$detected_patterns"
  fi

  # --- Wait for gh result ---
  if [[ -n "${GH_PID:-}" ]]; then
    if wait "$GH_PID" 2>/dev/null; then
      local gh_result
      gh_result=$(cat /tmp/harness-gh-result-$$ 2>/dev/null || echo "false")
      [[ "$gh_result" == "true" ]] && pr_created="true"
    else
      pr_check_timeout="true"
    fi
    rm -f /tmp/harness-gh-result-$$
  fi

  # --- Compute duration_seconds ---
  local duration_seconds="null"
  if [[ -f "$events_file" ]]; then
    local start_ts
    start_ts=$(jq -r "select(.event==\"session.start\" and .session_id==\"$SESSION_PREFIX\") | .ts" "$events_file" 2>/dev/null | tail -1)
    if [[ -n "$start_ts" && "$start_ts" != "null" ]]; then
      local start_epoch end_epoch
      start_epoch=$(date -j -f "%Y-%m-%dT%H:%M:%SZ" "$start_ts" +%s 2>/dev/null || date -d "$start_ts" +%s 2>/dev/null || echo "")
      end_epoch=$(date -u +%s)
      if [[ -n "$start_epoch" ]]; then
        duration_seconds=$((end_epoch - start_epoch))
      fi
    fi
  fi
  # Fallback: progress file creation time
  if [[ "$duration_seconds" == "null" && -f "$progress_file" ]]; then
    local file_epoch
    file_epoch=$(stat -f %B "$progress_file" 2>/dev/null || stat -c %W "$progress_file" 2>/dev/null || echo "")
    if [[ -n "$file_epoch" && "$file_epoch" != "0" ]]; then
      duration_seconds=$(( $(date -u +%s) - file_epoch ))
    fi
  fi

  # --- Derive heuristic_outcome ---
  local heuristic_outcome="partial"
  if [[ "$tests_passing" == "false" || "$constraint_violations_blocked" -gt 0 ]]; then
    heuristic_outcome="failed"
  elif [[ ("$pr_created" == "true" || "$progress_marked_complete" == "true") && ("$tests_passing" == "true" || "$tests_passing" == "null") && "$unresolved_loops" -eq 0 && "$constraint_violations_blocked" -eq 0 ]]; then
    heuristic_outcome="success"
  fi

  # --- Compute outcome_agreement ---
  local outcome_agreement="false"
  [[ "$agent_outcome" == "$heuristic_outcome" ]] && outcome_agreement="true"

  # --- Compute team_context ---
  local team_context="false"
  if [[ -f "$events_file" ]]; then
    local yesterday
    yesterday=$(date -u -v-1d +"%Y-%m-%dT%H:%M:%SZ" 2>/dev/null || date -u -d "yesterday" +"%Y-%m-%dT%H:%M:%SZ" 2>/dev/null || echo "1970-01-01T00:00:00Z")
    local team_count
    team_count=$(jq -r "select(.event | startswith(\"team.\")) | select(.branch==\"$BRANCH\" and .ts > \"$yesterday\")" "$events_file" 2>/dev/null | wc -l | tr -d ' ')
    [[ "$team_count" -gt 0 ]] && team_context="true"
  fi

  # --- Emit session.end ---
  emit_event "session.end" "$(jq -n -c \
    --arg mode "$mode" \
    --arg ao "$agent_outcome" \
    --arg ho "$heuristic_outcome" \
    --argjson oa "$outcome_agreement" \
    --argjson pr "$pr_created" \
    --argjson prt "$pr_check_timeout" \
    --argjson tp "$tests_passing" \
    --argjson ul "$unresolved_loops" \
    --argjson cvb "$constraint_violations_blocked" \
    --argjson pmc "$progress_marked_complete" \
    --argjson dur "$duration_seconds" \
    --argjson cc "$compaction_count" \
    --argjson lc "$loop_count" \
    --argjson tv "$total_violations" \
    --argjson sb "$semantic_blocks" \
    --argjson tc "$team_context" \
    '{
      mode:$mode, agent_outcome:$ao, heuristic_outcome:$ho, outcome_agreement:$oa,
      heuristic_signals:{pr_created:$pr, pr_check_timeout:$prt, tests_passing:$tp, unresolved_loops:$ul, constraint_violations_blocked:$cvb, progress_marked_complete:$pmc},
      duration_seconds:$dur, compaction_count:$cc, loop_count:$lc, total_violations:$tv, semantic_blocks:$sb, team_context:$tc
    }'
  )"
}

    echo "## Signals"
    echo "$session_end" | jq -r '.heuristic_signals | to_entries[] | "- \(.key): \(.value)"'
    echo "- Compactions: $cc"
    echo "- Semantic blocks: $(echo "$session_end" | jq -r '.semantic_blocks')"
  } > "$pm_file"

#!/usr/bin/env bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
source "$SCRIPT_DIR/lib.sh"

# Read hook input from stdin
INPUT=$(cat)
export SESSION_PREFIX=$(get_session_prefix "$INPUT")
export CWD=$(get_field "$INPUT" ".cwd")
export BRANCH=$(git -C "$CWD" rev-parse --abbrev-ref HEAD 2>/dev/null || echo "unknown")

TASK_ID=$(get_field "$INPUT" ".task_id")
[[ -z "$TASK_ID" ]] && TASK_ID="unknown"
EVENTS_FILE="$(get_analytics_dir "$CWD")/events.jsonl"

# Compute advisory signals
RECENT_BLOCKED=0
RECENT_WARN=0
RECENT_LOOPS=0

if [[ -f "$EVENTS_FILE" ]]; then
  # Time window: since last team.task_completed on this branch
  LAST_TC_TS=$(jq -s -r "[.[] | select(.event==\"team.task_completed\" and .branch==\"$BRANCH\")] | sort_by(.ts) | last | .ts // \"1970-01-01T00:00:00Z\"" "$EVENTS_FILE" 2>/dev/null)

  RECENT_BLOCKED=$(jq -r "select(.event==\"constraint.violation\" and .branch==\"$BRANCH\" and .decision==\"deny\" and .ts > \"$LAST_TC_TS\")" "$EVENTS_FILE" 2>/dev/null | wc -l | tr -d ' ')
  RECENT_WARN=$(jq -r "select(.event==\"constraint.violation\" and .branch==\"$BRANCH\" and .decision==\"allow\" and .ts > \"$LAST_TC_TS\")" "$EVENTS_FILE" 2>/dev/null | wc -l | tr -d ' ')
  RECENT_LOOPS=$(jq -r "select(.event==\"loop.detected\" and .branch==\"$BRANCH\" and .ts > \"$LAST_TC_TS\")" "$EVENTS_FILE" 2>/dev/null | wc -l | tr -d ' ')
fi

# Emit event with scope "team" via emit_event
emit_event "team.task_completed" "$(jq -n -c \
  --arg tid "$TASK_ID" \
  --argjson rb "$RECENT_BLOCKED" \
  --argjson rw "$RECENT_WARN" \
  --argjson rl "$RECENT_LOOPS" \
  '{task_id:$tid, advisory_signals:{recent_blocked_violations:$rb, recent_warn_violations:$rw, recent_loops_on_branch:$rl}}'
)" "team"

# Always exit 0
exit 0
